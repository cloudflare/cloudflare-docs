This example shows how to use React Email to build modern email templates with components and send them via Cloudflare Email Service.

```typescript
// First install: npm install react-email @react-email/components

import { render } from '@react-email/components';
import { Html, Head, Body, Container, Text, Button } from '@react-email/components';

interface Env {
  EMAIL: EmailBinding;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const { email, name } = await request.json();

    // Render React component to HTML
    const welcomeHtml = await render(WelcomeEmail({ name }));

    // Send email using Cloudflare Email Service
    await env.EMAIL.send({
      to: email,
      from: 'welcome@example.com',
      subject: 'Welcome to our platform!',
      html: welcomeHtml
    });

    return new Response(JSON.stringify({ success: true }));
  }
};

// React Email Component
function WelcomeEmail({ name }: { name: string }) {
  return (
    <Html>
      <Head />
      <Body style={bodyStyle}>
        <Container style={containerStyle}>
          <Text style={headingStyle}>
            Welcome {name}! 🎉
          </Text>
          
          <Text style={textStyle}>
            We're excited to have you on board. Click the button below to get started:
          </Text>

          <Button style={buttonStyle} href="https://app.example.com/dashboard">
            Get Started
          </Button>

          <Text style={footerStyle}>
            Questions? Reply to this email or contact support@example.com
          </Text>
        </Container>
      </Body>
    </Html>
  );
}

// Styles
const bodyStyle = {
  backgroundColor: '#f6f9fc',
  fontFamily: '-apple-system,BlinkMacSystemFont,"Segoe UI",Roboto,sans-serif',
};

const containerStyle = {
  backgroundColor: '#ffffff',
  margin: '0 auto',
  padding: '40px 20px',
  maxWidth: '600px',
};

const headingStyle = {
  fontSize: '24px',
  fontWeight: '600',
  color: '#333',
  textAlign: 'center' as const,
  margin: '0 0 20px 0',
};

const textStyle = {
  fontSize: '16px',
  lineHeight: '24px',
  color: '#666',
  margin: '0 0 30px 0',
};

const buttonStyle = {
  backgroundColor: '#007ee6',
  borderRadius: '4px',
  color: '#fff',
  fontSize: '16px',
  fontWeight: '600',
  textDecoration: 'none',
  textAlign: 'center' as const,
  padding: '12px 20px',
  margin: '0 auto 30px',
  display: 'block',
  width: 'fit-content',
};

const footerStyle = {
  color: '#999',
  fontSize: '14px',
  textAlign: 'center' as const,
  margin: '20px 0 0 0',
};
```
