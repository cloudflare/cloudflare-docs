## Properties

| Property | Type | Required | Default | Description |
|----------|------|----------|---------|-------------|
| `meeting` | `Meeting` | ✅ | - | Meeting object |

## Usage Examples

### Basic Usage

```tsx
import { RtkWaitingScreen } from '@cloudflare/realtimekit-react-ui';

function MyComponent() {
  return <RtkWaitingScreen />;
}
```

### With Properties

```tsx
import { RtkWaitingScreen } from '@cloudflare/realtimekit-react-ui';

function MyComponent() {
  return (
    <RtkWaitingScreen
      meeting={meeting}
    />
  );
}
```
