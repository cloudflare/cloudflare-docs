name: Wrangler Deployment

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  validate-and-deploy:
    runs-on: ubuntu-latest
    env:
      WRANGLER_API_TOKEN: ${{ secrets.WRANGLER_API_TOKEN }}
      ALLOW_WRANGLER_OVERRIDE: "false"  # governance override flag
    steps:
      - name: Checkout repository
        uses: actions/checkout@v3

      - name: Set up Node.js
        uses: actions/setup-node@v3
        with:
          node-version: 20

      - name: Install dependencies
        run: npm ci

      - name: Validate Wrangler version
        run: bash validate-wrangler.sh

      - name: Deploy via Wrangler
        run: node deploy.js
