enum RolUsuario { agente, almacen, admin }

class Usuario {
  final String nombre;
  final RolUsuario rol;
  final String password;

  Usuario({required this.nombre, required this.rol, required this.password});
}

class Producto {
  final String codigo;
  final String nombre;
  double precio;
  int stock;

  Producto({
    required this.codigo,
    required this.nombre,
    required this.precio,
    required this.stock,
  });
}

class OrderItem {
  final String productCode;
  final String productName;
  int quantity;

  OrderItem({required this.productCode, required this.productName, required this.quantity});
}

class PedidoDocumento {
  final String id;
  final String cliente;
  final String agenteNombre;
  String contenidoAgente; // The raw, editable "file" content from agent
  String contenidoAlmacen; // The content marked by Almacén (✅/❌)
  String estadoDocumento; // "Pendiente de Recepción (Impresora02)", "Enviado a Almacén", "Procesado (Completo)", "Procesado (Incompleto)"
  String observacionAlmacen;
  final DateTime fechaCreacion;
  DateTime? fechaEnvioAlmacen; // When it was moved from Impresora02 queue to "sent to warehouse"
  DateTime? fechaProcesamientoAlmacen;

  PedidoDocumento({
    required this.id,
    required this.cliente,
    required this.agenteNombre,
    required this.contenidoAgente,
    required this.fechaCreacion,
    this.contenidoAlmacen = "", // Initialize as empty
    this.estadoDocumento = "Pendiente de Recepción (Impresora02)", // Initial state when created by agent
    this.observacionAlmacen = "",
    this.fechaEnvioAlmacen,
    this.fechaProcesamientoAlmacen,
  });

  // Constructor para copiar y modificar
  PedidoDocumento copyWith({
    String? id,
    String? cliente,
    String? agenteNombre,
    String? contenidoAgente,
    String? contenidoAlmacen,
    String? estadoDocumento,
    String? observacionAlmacen,
    DateTime? fechaCreacion,
    DateTime? fechaEnvioAlmacen,
    DateTime? fechaProcesamientoAlmacen,
  }) {
    return PedidoDocumento(
      id: id ?? this.id,
      cliente: cliente ?? this.cliente,
      agenteNombre: agenteNombre ?? this.agenteNombre,
      contenidoAgente: contenidoAgente ?? this.contenidoAgente,
      contenidoAlmacen: contenidoAlmacen ?? this.contenidoAlmacen,
      estadoDocumento: estadoDocumento ?? this.estadoDocumento,
      observacionAlmacen: observacionAlmacen ?? this.observacionAlmacen,
      fechaCreacion: fechaCreacion ?? this.fechaCreacion,
      fechaEnvioAlmacen: fechaEnvioAlmacen ?? this.fechaEnvioAlmacen,
      fechaProcesamientoAlmacen: fechaProcesamientoAlmacen ?? this.fechaProcesamientoAlmacen,
    );
  }
}

class Notificacion {
  final String mensaje;
  final DateTime fecha;

  Notificacion({
    required this.mensaje,
    required this.fecha,
  });
}

class MensajeChat {
  final String sender;
  final String message;
  final DateTime timestamp;

  MensajeChat({
    required this.sender,
    required this.message,
    required this.timestamp,
  });
}

class SistemaCentral extends ChangeNotifier {
  List<Producto> inventario = <Producto>[
    Producto(codigo: "CUA001", nombre: "Cuadernos A4", precio: 25.0, stock: 50),
    Producto(codigo: "LAP002", nombre: "Lapiceros Azules", precio: 5.0, stock: 30),
    Producto(codigo: "CAL003", nombre: "Calculadora Científica", precio: 120.0, stock: 15),
    Producto(codigo: "RES004", nombre: "Resaltadores Pack x4", precio: 18.0, stock: 25),
    Producto(codigo: "ENG005", nombre: "Engrapadora Pequeña", precio: 35.0, stock: 10),
  ];

  final Map<String, Usuario> _users = <String, Usuario>{
    "Admin": Usuario(nombre: "Admin", rol: RolUsuario.admin, password: "Flamingo02"),
    "Almacén": Usuario(nombre: "Almacén", rol: RolUsuario.almacen, password: "warehouse_pass"),
    "V264": Usuario(nombre: "V264", rol: RolUsuario.agente, password: "agent_pass"),
    "MIG": Usuario(nombre: "MIG", rol: RolUsuario.agente, password: "agent_pass"),
    "PRC": Usuario(nombre: "PRC", rol: RolUsuario.agente, password: "agent_pass"),
    "GAB": Usuario(nombre: "GAB", rol: RolUsuario.agente, password: "agent_pass"),
  };

  final Map<String, bool> _userImpresoraStates = <String, bool>{
    "Admin": false,
    "Almacén": false,
    "V264": false,
    "MIG": false,
    "PRC": false,
    "GAB": false,
  };

  bool getImpresoraState(String userName) => _userImpresoraStates[userName] ?? false;

  void setImpresoraState(String userName, bool isEnabled) {
    _userImpresoraStates[userName] = isEnabled;
    notificaciones.insert(
      0,
      Notificacion(
        mensaje: "'$userName' ha ${isEnabled ? 'activado' : 'desactivado'} su Impresora02.",
        fecha: DateTime.now(),
      ),
    );
    notifyListeners();
  }

  List<Usuario> get allUsers => _users.values.toList();
  List<Usuario> get agents => _users.values.where((Usuario u) => u.rol == RolUsuario.agente).toList();
  Usuario? getUser(String name) => _users[name];

  void addUsuario(Usuario user) {
    if (_users.containsKey(user.nombre)) {
      return;
    }
    _users[user.nombre] = user;
    _userImpresoraStates[user.nombre] = false; // Initialize Impresora02 state for new user
    notificaciones.insert(
      0,
      Notificacion(
        mensaje: "Nuevo usuario '${user.nombre}' (${user.rol.name}) registrado.",
        fecha: DateTime.now(),
      ),
    );
    notifyListeners();
  }

  void removeUsuario(String nombre) {
    if (nombre == "Admin" || nombre == "Almacén") {
      return;
    }

    if (_users.remove(nombre) != null) {
      _privateChatMessages.remove(nombre);
      _userImpresoraStates.remove(nombre); // Remove Impresora02 state
      notificaciones.insert(
        0,
        Notificacion(
          mensaje: "Usuario '$nombre' eliminado.",
          fecha: DateTime.now(),
        ),
      );
      documentosPedido.removeWhere((PedidoDocumento doc) => doc.agenteNombre == nombre);
      _impresoraQueue.removeWhere((PedidoDocumento doc) => doc.agenteNombre == nombre);
      notifyListeners();
    }
  }

  void updateUsuarioPassword(String nombre, String newPassword) {
    if (_users.containsKey(nombre)) {
      final Usuario? oldUser = _users[nombre];
      if (oldUser != null) {
        _users[nombre] = Usuario(nombre: nombre, rol: oldUser.rol, password: newPassword);
        notificaciones.insert(
          0,
          Notificacion(
            mensaje: "Contraseña de usuario '$nombre' actualizada.",
            fecha: DateTime.now(),
          ),
        );
        notifyListeners();
      }
    }
  }

  List<PedidoDocumento> documentosPedido = <PedidoDocumento>[]; // Documents already received by warehouse
  List<PedidoDocumento> _impresoraQueue = <PedidoDocumento>[]; // Documents pending reception by warehouse
  List<PedidoDocumento> get impresoraQueue => _impresoraQueue;

  int _nextPedidoDocumentoId = 1; // Starting from D001

  List<Notificacion> notificaciones = <Notificacion>[
    Notificacion(
      mensaje: "Bienvenido al sistema.",
      fecha: DateTime.now().subtract(const Duration(minutes: 5)),
    ),
  ];

  List<MensajeChat> publicChatMessages = <MensajeChat>[
    MensajeChat(
      sender: "Almacén",
      message: "Bienvenidos al chat público. ¡Esperamos sus consultas!",
      timestamp: DateTime.now().subtract(const Duration(minutes: 10)),
    ),
  ];

  final Map<String, List<MensajeChat>> _privateChatMessages = <String, List<MensajeChat>>{};

  List<MensajeChat> getPrivateChatForAgent(String agentName) {
    return _privateChatMessages[agentName] ?? <MensajeChat>[];
  }

  void addPublicChatMessage(String sender, String message) {
    publicChatMessages.insert(0, MensajeChat(sender: sender, message: message, timestamp: DateTime.now()));
    notifyListeners();
  }

  void addPrivateChatMessage(String chatAgentName, String sender, String message) {
    _privateChatMessages.putIfAbsent(chatAgentName, () => <MensajeChat>[]).insert(0, MensajeChat(sender: sender, message: message, timestamp: DateTime.now()));
    notificaciones.insert(0, Notificacion(mensaje: "Nuevo mensaje privado para '$chatAgentName' de '$sender'.", fecha: DateTime.now()));
    notifyListeners();
  }

  Producto? getProductoByCodeOrName(String query) {
    try {
      return inventario.firstWhere(
          (Producto p) => p.nombre.toLowerCase() == query.toLowerCase() || p.codigo.toLowerCase() == query.toLowerCase());
    } catch (e) {
      return null;
    }
  }

  void addProducto(String codigo, String nombre, double precio, int stock) {
    if (inventario.any((Producto p) =>
        p.codigo.toLowerCase() == codigo.toLowerCase() ||
        p.nombre.toLowerCase() == nombre.toLowerCase())) {
      return;
    }
    inventario.add(Producto(codigo: codigo, nombre: nombre, precio: precio, stock: stock));
    notificaciones.insert(
      0,
      Notificacion(
        mensaje: "Nuevo producto '$nombre' (Código: $codigo) agregado al inventario.",
        fecha: DateTime.now(),
      ),
    );
    notifyListeners();
  }

  void updateProducto(String codigo, {double? newPrecio, int? newStock}) {
    final int index = inventario.indexWhere((Producto p) => p.codigo == codigo);
    if (index != -1) {
      if (newPrecio != null) {
        inventario[index].precio = newPrecio;
      }
      if (newStock != null) {
        inventario[index].stock = newStock;
      }
      notificaciones.insert(
        0,
        Notificacion(
          mensaje:
              "Producto '${inventario[index].nombre}' (Código: $codigo) actualizado. Precio: \$${inventario[index].precio.toStringAsFixed(2)}, Stock: ${inventario[index].stock}.",
          fecha: DateTime.now(),
        ),
      );
      notifyListeners();
    }
  }

  void deleteProducto(String codigo) {
    final int initialLength = inventario.length;
    inventario.removeWhere((Producto p) => p.codigo == codigo);
    if (inventario.length < initialLength) {
      notificaciones.insert(
        0,
        Notificacion(
          mensaje: "Producto con código '$codigo' eliminado del inventario.",
          fecha: DateTime.now(),
        ),
      );
      notifyListeners();
    }
  }

  void crearPedidoDocumento(String cliente, List<OrderItem> items, String agenteNombre) {
    if (cliente.isEmpty || items.isEmpty || agenteNombre.isEmpty) return;
    if (!getImpresoraState(agenteNombre)) {
      // Should be checked in UI, but a safeguard
      return;
    }

    final String newId = "D${_nextPedidoDocumentoId.toString().padLeft(3, '0')}";
    _nextPedidoDocumentoId++;

    final StringBuffer documentContentBuffer = StringBuffer();
    documentContentBuffer.writeln("Pedido para Cliente: $cliente");
    documentContentBuffer.writeln("Agente: $agenteNombre");
    documentContentBuffer.writeln("Fecha: ${DateTime.now().toLocal().toString().split('.')[0]}");
    documentContentBuffer.writeln("--- ITEMS ---");
    for (final OrderItem item in items) {
      final Producto productInInventory = inventario.firstWhere((Producto p) => p.codigo == item.productCode);
      documentContentBuffer.writeln("- ${item.productName} (Código: ${item.productCode}) x ${item.quantity} unidades. Precio Unitario: \$${productInInventory.precio.toStringAsFixed(2)}");
    }
    documentContentBuffer.writeln("--- FIN DE PEDIDO ---");
    final String contenidoAgente = documentContentBuffer.toString();

    final PedidoDocumento newDocumento = PedidoDocumento(
      id: newId,
      cliente: cliente,
      agenteNombre: agenteNombre,
      contenidoAgente: contenidoAgente,
      fechaCreacion: DateTime.now(),
      estadoDocumento: "Pendiente de Recepción (Impresora02)", // New initial state
      fechaEnvioAlmacen: null, // Not yet "sent to warehouse"
    );
    _impresoraQueue.insert(0, newDocumento); // Add to Impresora02 queue

    notificaciones.insert(
      0,
      Notificacion(
        mensaje:
            "Pedido $newId para el cliente '$cliente' de '$agenteNombre' enviado a la cola de impresión (Impresora02).",
        fecha: DateTime.now(),
      ),
    );

    notifyListeners();
  }

  void receiveDocumentByAlmacen(String documentId) {
    final int index = _impresoraQueue.indexWhere((PedidoDocumento d) => d.id == documentId);
    if (index != -1) {
      final PedidoDocumento doc = _impresoraQueue.removeAt(index);
      final PedidoDocumento receivedDoc = doc.copyWith(
        estadoDocumento: "Enviado a Almacén",
        fechaEnvioAlmacen: DateTime.now(),
      );
      documentosPedido.insert(0, receivedDoc); // Move to documents for processing

      // Notification for Almacén (general system log)
      notificaciones.insert(
        0,
        Notificacion(
          mensaje:
              "Documento de pedido ${doc.id} de '${doc.agenteNombre}' para cliente '${doc.cliente}' ha sido recibido por Almacén (sim. Impresora02).",
          fecha: DateTime.now(),
        ),
      );
      // Specific notification for the agent
      notificaciones.insert(
        0,
        Notificacion(
          mensaje: "Tu pedido ${doc.id} para '${doc.cliente}' ha sido recibido por Almacén.",
          fecha: DateTime.now(),
        ),
      );
      notifyListeners();
    }
  }

  void procesarPedidoDocumentoAlmacen(
      String documentId, String contenidoMarcado, String estadoFinal, String observacion) {
    final int index = documentosPedido.indexWhere((PedidoDocumento d) => d.id == documentId);
    if (index != -1) {
      final PedidoDocumento doc = documentosPedido[index];
      documentosPedido[index] = doc.copyWith(
        contenidoAlmacen: contenidoMarcado,
        estadoDocumento: estadoFinal,
        observacionAlmacen: observacion,
        fechaProcesamientoAlmacen: DateTime.now(),
      );

      notificaciones.insert(
        0,
        Notificacion(
          mensaje:
              "Tu pedido ${doc.id} para '${doc.cliente}' ha sido procesado por Almacén ($estadoFinal). Observación: ${observacion.isNotEmpty ? observacion : 'Ninguna'}",
          fecha: DateTime.now(),
        ),
      );
      notifyListeners();
    }
  }

  List<PedidoDocumento> getDocumentosPendientesAlmacen() {
    return documentosPedido
        .where((PedidoDocumento doc) => doc.estadoDocumento == "Enviado a Almacén")
        .toList();
  }

  List<PedidoDocumento> getDocumentosByAgente(String agenteNombre) {
    return documentosPedido
        .where((PedidoDocumento doc) => doc.agenteNombre == agenteNombre)
        .toList();
  }
}

Color getEstadoDocumentoColor(String estado) {
  switch (estado) {
    case "Pendiente de Recepción (Impresora02)":
      return Colors.blueGrey;
    case "Enviado a Almacén":
      return Colors.orange;
    case "Procesado (Completo)":
      return Colors.green;
    case "Procesado (Incompleto)":
      return Colors.red;
    default:
      return Colors.grey;
  }
}

class LoginSimulado extends StatelessWidget {
  LoginSimulado({super.key});

  void _showPasswordDialog(BuildContext context, Usuario user) {
    final TextEditingController passwordController = TextEditingController();
    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: Text("Ingresar como ${user.nombre}"),
          content: TextField(
            controller: passwordController,
            obscureText: true,
            decoration: const InputDecoration(labelText: "Contraseña"),
          ),
          actions: <Widget>[
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text("Cancelar"),
            ),
            ElevatedButton(
              onPressed: () {
                final String enteredPassword = passwordController.text;
                final SistemaCentral system =
                    Provider.of<SistemaCentral>(dialogContext, listen: false);
                final Usuario? latestUser =
                    system.getUser(user.nombre);

                if (latestUser != null && enteredPassword == latestUser.password) {
                  Navigator.pop(dialogContext);
                  Navigator.push(
                    context,
                    MaterialPageRoute<void>(
                      builder: (BuildContext _) => PantallaSegunRol(usuario: latestUser),
                    ),
                  );
                } else {
                  ScaffoldMessenger.of(dialogContext).showSnackBar(
                    const SnackBar(content: Text("Contraseña incorrecta o usuario no encontrado.")),
                  );
                }
              },
              child: const Text("Ingresar"),
            ),
          ],
        );
      },
    ).then((_) {
      passwordController.dispose();
    });
  }

  @override
  Widget build(BuildContext context) {
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context);
    final List<Usuario> agents = sistema.agents;
    final Usuario? adminUser = sistema.getUser("Admin");
    final Usuario? almacenUser = sistema.getUser("Almacén");

    return Scaffold(
      appBar: AppBar(title: const Text("Seleccionar Rol")),
      body: Center(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(16.0),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: <Widget>[
              ...agents.map<Widget>((Usuario agent) => Padding(
                    padding: const EdgeInsets.only(bottom: 16),
                    child: ElevatedButton(
                      onPressed: () => _showPasswordDialog(context, agent),
                      child: Text("Entrar como Agente ${agent.nombre}"),
                    ),
                  )).toList(),
              if (almacenUser != null)
                Padding(
                  padding: const EdgeInsets.only(bottom: 16),
                  child: ElevatedButton(
                    onPressed: () => _showPasswordDialog(context, almacenUser),
                    child: const Text("Entrar como Almacén"),
                  ),
                ),
              if (adminUser != null)
                ElevatedButton(
                  onPressed: () => _showPasswordDialog(context, adminUser),
                  child: const Text("Entrar como Administrador"),
                ),
            ],
          ),
        ),
      ),
    );
  }
}

class PantallaSegunRol extends StatelessWidget {
  final Usuario usuario;

  const PantallaSegunRol({required this.usuario, super.key});

  @override
  Widget build(BuildContext context) {
    switch (usuario.rol) {
      case RolUsuario.agente:
        return PanelAgente(usuario: usuario);
      case RolUsuario.almacen:
        return PanelAlmacen(usuario: usuario);
      case RolUsuario.admin:
        return PanelAdmin(usuario: usuario);
    }
  }
}

class _AlmacenImpresoraReceptionTab extends StatelessWidget {
  final Usuario usuario;
  const _AlmacenImpresoraReceptionTab({required this.usuario, super.key});

  void _receiveDocument(BuildContext context, String documentId) {
    Provider.of<SistemaCentral>(context, listen: false).receiveDocumentByAlmacen(documentId);
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text("Documento $documentId recibido de la Impresora02."),
        backgroundColor: Colors.green,
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context);
    final bool impresoraEnabled = sistema.getImpresoraState(usuario.nombre);
    final List<PedidoDocumento> impresoraQueue = sistema.impresoraQueue;

    return Column(
      children: <Widget>[
        Card(
          margin: const EdgeInsets.all(16),
          child: SwitchListTile(
            title: const Text("Impresora02 del Almacén"),
            subtitle: Text(
              impresoraEnabled
                  ? "Recepción de documentos de pedido activa."
                  : "Recepción de documentos de pedido inactiva.",
            ),
            value: impresoraEnabled,
            onChanged: (bool value) {
              sistema.setImpresoraState(usuario.nombre, value);
            },
            secondary: const Icon(Icons.print),
          ),
        ),
        if (!impresoraEnabled)
          const Expanded(
            child: Center(
              child: Padding(
                padding: EdgeInsets.all(16.0),
                child: Text(
                  "Active la Impresora02 para recibir documentos de pedido enviados por los agentes.",
                  textAlign: TextAlign.center,
                  style: TextStyle(fontStyle: FontStyle.italic, color: Colors.grey),
                ),
              ),
            ),
          )
        else if (impresoraQueue.isEmpty)
          const Expanded(
            child: Center(
              child: Padding(
                padding: EdgeInsets.all(16.0),
                child: Text(
                  "No hay documentos pendientes en la cola de recepción de Impresora02.",
                  textAlign: TextAlign.center,
                  style: TextStyle(fontStyle: FontStyle.italic, color: Colors.grey),
                ),
              ),
            ),
          )
        else
          Expanded(
            child: ListView.builder(
              padding: const EdgeInsets.symmetric(horizontal: 16),
              itemCount: impresoraQueue.length,
              itemBuilder: (BuildContext context, int index) {
                final PedidoDocumento document = impresoraQueue[index];
                return Card(
                  margin: const EdgeInsets.only(bottom: 8),
                  elevation: 2,
                  child: ListTile(
                    leading: const Icon(Icons.print, color: Colors.blueGrey),

class _PedidoDocumentosAlmacenTabState extends State<_PedidoDocumentosAlmacenTab> {

  final Set<String> _notifiedDocumentIds = <String>{};

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context, listen: false);
    final List<PedidoDocumento> pendingDocs = sistema.getDocumentosPendientesAlmacen();

    final List<PedidoDocumento> newPendingDocs = pendingDocs
        .where((PedidoDocumento doc) => !_notifiedDocumentIds.contains(doc.id))
        .toList();

    if (newPendingDocs.isNotEmpty) {
      WidgetsBinding.instance.addPostFrameCallback((_) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
                "¡NUEVO PEDIDO! Tienes ${newPendingDocs.length} documento(s) pendiente(s) de Almacén."),
            duration: const Duration(seconds: 3),
            backgroundColor: Colors.deepOrange,
          ),
        );
        for (final PedidoDocumento doc in newPendingDocs) {
          _notifiedDocumentIds.add(doc.id);
        }
      });
    }

class _CrearPedidoDocumentoTab extends StatefulWidget {

class _CrearPedidoDocumentoTabState extends State<_CrearPedidoDocumentoTab> {
  final TextEditingController _clienteController = TextEditingController();
  final Map<String, OrderItem> _currentOrderItems = <String, OrderItem>{};

  @override
  void dispose() {
    _clienteController.dispose();
    super.dispose();
  }

  void _addOrUpdateItemInOrder(Producto product, int quantity) {
    setState(() {
      _currentOrderItems[product.codigo] = OrderItem(
        productCode: product.codigo,
        productName: product.nombre,
        quantity: quantity,
      );
    });
  }

  void _removeItemFromOrder(String productCode) {
    setState(() {
      _currentOrderItems.remove(productCode);
    });
  }

  void _showAddProductToOrderDialog(BuildContext context) {
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context, listen: false);
    Producto? selectedProduct;
    final TextEditingController quantityController = TextEditingController(text: "1");

    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return StatefulBuilder(
          builder: (BuildContext innerContext, void Function(void Function()) setInnerState) {
            return AlertDialog(
              title: const Text("Agregar Producto al Pedido"),
              content: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: <Widget>[
                    DropdownButtonFormField<Producto>(
                      decoration: const InputDecoration(labelText: "Producto"),
                      initialValue: selectedProduct,
                      items: sistema.inventario.map<DropdownMenuItem<Producto>>((Producto product) {
                        return DropdownMenuItem<Producto>(
                          value: product,
                          child: Text("${product.nombre} (\$${product.precio.toStringAsFixed(2)})"),
                        );
                      }).toList(),
                      onChanged: (Producto? newValue) {
                        setInnerState(() {
                          selectedProduct = newValue;
                        });
                      },
                    ),
                    const SizedBox(height: 10),
                    TextField(
                      controller: quantityController,
                      decoration: InputDecoration(
                        labelText: "Cantidad",
                        hintText: selectedProduct != null ? "Stock: ${selectedProduct!.stock}" : "",
                      ),
                      keyboardType: TextInputType.number,
                    ),
                  ],
                ),
              ),
              actions: <Widget>[
                TextButton(
                  onPressed: () => Navigator.pop(innerContext),
                  child: const Text("Cancelar"),
                ),
                ElevatedButton(
                  onPressed: () {
                    final int? quantity = int.tryParse(quantityController.text.trim());
                    if (selectedProduct == null || quantity == null || quantity <= 0) {
                      ScaffoldMessenger.of(innerContext).showSnackBar(
                        const SnackBar(content: Text("Por favor, seleccione un producto y una cantidad válida.")),
                      );
                      return;
                    }
                    if (quantity > selectedProduct!.stock) {
                      ScaffoldMessenger.of(innerContext).showSnackBar(
                        SnackBar(content: Text("Cantidad excede el stock disponible (${selectedProduct!.stock}).")),
                      );
                      return;
                    }
                    _addOrUpdateItemInOrder(selectedProduct!, quantity);
                    Navigator.pop(innerContext);
                  },
                  child: const Text("Agregar"),
                ),
              ],
            );
          },
        );
      },
    ).then((_) {
      quantityController.dispose();
    });
  }

  void _submitDocument() {
    final String cliente = _clienteController.text.trim();
    final List<OrderItem> items = _currentOrderItems.values.toList();
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context, listen: false);

    if (!sistema.getImpresoraState(widget.usuario.nombre)) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("Debe activar la Impresora02 para enviar pedidos.")),
      );
      return;
    }

    if (cliente.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("Por favor, ingrese el nombre del cliente.")),
      );
      return;
    }
    if (items.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("El pedido no puede estar vacío. Agregue al menos un producto.")),
      );
      return;
    }

    sistema.crearPedidoDocumento(cliente, items, widget.usuario.nombre);

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
          content: Text("Pedido para $cliente enviado a la cola de impresión Impresora02.")),
    );

    _clienteController.clear();
    setState(() {
      _currentOrderItems.clear();
    });
  }

      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: <Widget>[

            controller: _clienteController,

          ),

          _currentOrderItems.isEmpty
              ? const Padding(
                  padding: EdgeInsets.symmetric(vertical: 20.0),
                  child: Center(
                      child: Text("No hay productos en el pedido. Agregue algunos.",
                          style: TextStyle(fontStyle: FontStyle.italic))),
                )
              : ListView.builder(
                  shrinkWrap: true,
                  physics: const NeverScrollableScrollPhysics(),
                  itemCount: _currentOrderItems.length,
                  itemBuilder: (BuildContext context, int index) {
                    final OrderItem item = _currentOrderItems.values.toList()[index];
                    return Card(
                      margin: const EdgeInsets.only(bottom: 8),
                      elevation: 1,
                      child: ListTile(
                        title: Text(item.productName),
                        subtitle: Text("Código: ${item.productCode} - Cantidad: ${item.quantity}"),
                        trailing: IconButton(
                          icon: const Icon(Icons.delete, color: Colors.red),
                          onPressed: () => _removeItemFromOrder(item.productCode),
                        ),
                      ),
                    );
                  },
                ),
          const SizedBox(height: 20),
          Center(
            child: ElevatedButton.icon(
              style: ElevatedButton.styleFrom(
                backgroundColor: impresoraEnabled ? Colors.deepPurple : Colors.grey,
                padding: const EdgeInsets.symmetric(horizontal: 40, vertical: 15),
                textStyle: const TextStyle(fontSize: 18),
              ),
              onPressed: impresoraEnabled ? _submitDocument : null,
              icon: const Icon(Icons.print),

                child: Text(
                  "Active su Impresora02 para enviar el pedido.",

class _ChatMessagesView extends StatelessWidget {
  final Usuario usuario;
  final List<MensajeChat> messages;
  final ScrollController scrollController;

  const _ChatMessagesView({
    required this.usuario,
    required this.messages,
    required this.scrollController,
    super.key,
  });

  @override
  Widget build(BuildContext context) {
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (scrollController.hasClients) {
        scrollController.animateTo(
          scrollController.position.minScrollExtent,
          duration: const Duration(milliseconds: 300),
          curve: Curves.easeOut,
        );
      }
    });

    return ListView.builder(
      reverse: true,
      controller: scrollController,
      padding: const EdgeInsets.all(8),
      itemCount: messages.length,
      itemBuilder: (BuildContext context, int index) {
        final MensajeChat message = messages[index];
        final bool isMe = message.sender == usuario.nombre;

        return Align(
          alignment: isMe ? Alignment.centerRight : Alignment.centerLeft,
          child: Container(
            margin: const EdgeInsets.symmetric(vertical: 4, horizontal: 8),
            padding: const EdgeInsets.all(12),
            decoration: BoxDecoration(
              color: isMe ? Colors.deepPurple.shade100 : Colors.grey.shade200,
              borderRadius: BorderRadius.circular(12),
            ),
            child: Column(
              crossAxisAlignment: isMe ? CrossAxisAlignment.end : CrossAxisAlignment.start,
              children: <Widget>[
                Text(
                  isMe ? "Tú" : message.sender,
                  style: const TextStyle(fontWeight: FontWeight.bold),
                ),
                const SizedBox(height: 4),
                Text(message.message),
                const SizedBox(height: 4),
                Text(
                  "${message.timestamp.hour}:${message.timestamp.minute.toString().padLeft(2, '0')}",
                  style: const TextStyle(fontSize: 10, color: Colors.black54),
                ),
              ],
            ),
          ),
        );
      },
    );
  }
}

class _InventarioEditorTabState extends State<_InventarioEditorTab> {
  final TextEditingController _addCodeController = TextEditingController();
  final TextEditingController _addNameController = TextEditingController();
  final TextEditingController _addPriceController = TextEditingController();
  final TextEditingController _addStockController = TextEditingController();

  @override
  void dispose() {
    _addCodeController.dispose();
    _addNameController.dispose();
    _addPriceController.dispose();
    _addStockController.dispose();
    super.dispose();
  }

  void _showAddProductDialog(BuildContext context) {
    _addCodeController.clear();
    _addNameController.clear();
    _addPriceController.clear();
    _addStockController.clear();

    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: const Text("Agregar Nuevo Producto"),
          content: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: <Widget>[
                TextField(
                  controller: _addCodeController,
                  decoration: const InputDecoration(labelText: "Código del Producto (Ej: CUA001)"),
                ),
                const SizedBox(height: 10),
                TextField(
                  controller: _addNameController,
                  decoration: const InputDecoration(labelText: "Nombre del Producto"),
                ),
                const SizedBox(height: 10),
                TextField(
                  controller: _addPriceController,
                  decoration: const InputDecoration(labelText: "Precio"),
                  keyboardType: const TextInputType.numberWithOptions(decimal: true),
                ),
                const SizedBox(height: 10),
                TextField(
                  controller: _addStockController,
                  decoration: const InputDecoration(labelText: "Stock Inicial"),
                  keyboardType: TextInputType.number,
                ),
              ],
            ),
          ),
          actions: <Widget>[
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text("Cancelar"),
            ),
            ElevatedButton(
              onPressed: () {
                final String code = _addCodeController.text.trim();
                final String name = _addNameController.text.trim();
                final double? price = double.tryParse(_addPriceController.text.trim());
                final int? stock = int.tryParse(_addStockController.text.trim());

                if (code.isEmpty || name.isEmpty || price == null || stock == null || price < 0 || stock < 0) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(content: Text("Por favor, ingrese datos válidos.")),
                  );
                  return;
                }

                final SistemaCentral sistema = Provider.of<SistemaCentral>(context, listen: false);
                if (sistema.inventario.any(
                    (Producto p) => p.codigo.toLowerCase() == code.toLowerCase())) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(content: Text("Ya existe un producto con este código.")),
                  );
                  return;
                }
                if (sistema.inventario.any(
                    (Producto p) => p.nombre.toLowerCase() == name.toLowerCase())) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(content: Text("Ya existe un producto con este nombre.")),
                  );
                  return;
                }

                sistema.addProducto(code, name, price, stock);
                Navigator.pop(dialogContext);
                ScaffoldMessenger.of(context).showSnackBar(
                  SnackBar(content: Text("Producto '$name' (Código: $code) agregado con éxito.")),
                );
              },
              child: const Text("Agregar"),
            ),
          ],
        );
      },
    );
  }

  void _showEditProductDialog(BuildContext context, Producto product) {
    final TextEditingController editPriceController =
        TextEditingController(text: product.precio.toStringAsFixed(2));
    final TextEditingController editStockController =
        TextEditingController(text: product.stock.toString());

    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: Text("Editar Producto: ${product.nombre} (Código: ${product.codigo})"),
          content: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: <Widget>[
                TextField(
                  controller: editPriceController,
                  decoration: const InputDecoration(labelText: "Precio"),
                  keyboardType: const TextInputType.numberWithOptions(decimal: true),
                ),
                const SizedBox(height: 10),
                TextField(
                  controller: editStockController,
                  decoration: const InputDecoration(labelText: "Stock"),
                  keyboardType: TextInputType.number,
                ),
              ],
            ),
          ),
          actions: <Widget>[
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text("Cancelar"),
            ),
            ElevatedButton(
              onPressed: () {
                final double? newPrice = double.tryParse(editPriceController.text.trim());
                final int? newStock = int.tryParse(editStockController.text.trim());

                if (newPrice == null || newStock == null || newPrice < 0 || newStock < 0) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(content: Text("Por favor, ingrese valores válidos.")),
                  );
                  return;
                }

                final SistemaCentral sistema = Provider.of<SistemaCentral>(context, listen: false);
                sistema.updateProducto(product.codigo, newPrecio: newPrice, newStock: newStock);
                Navigator.pop(dialogContext);
                ScaffoldMessenger.of(context).showSnackBar(
                  SnackBar(content: Text("Producto '${product.nombre}' actualizado.")),
                );
              },
              child: const Text("Guardar"),
            ),
          ],
        );
      },
    ).then((_) {
      editPriceController.dispose();
      editStockController.dispose();
    });
  }

  @override
  Widget build(BuildContext context) {
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context);

    return Scaffold(
      body: ListView.builder(
        padding: const EdgeInsets.all(16),

          final Producto producto = sistema.inventario[index];
          return Card(
            margin: const EdgeInsets.only(bottom: 8),
            elevation: 2,
            child: ListTile(
              leading: const Icon(Icons.inventory, color: Colors.teal),
              title: Text(producto.nombre, style: const TextStyle(fontWeight: FontWeight.bold)),
              subtitle: Text(
                "Código: ${producto.codigo} - Precio: \$${producto.precio.toStringAsFixed(2)} - Stock: ${producto.stock}",
              ),
              trailing: Row(
                mainAxisSize: MainAxisSize.min,

                  IconButton(
                    icon: const Icon(Icons.edit, color: Colors.blue),
                    tooltip: "Editar Producto",
                    onPressed: () => _showEditProductDialog(context, producto),
                  ),
                  IconButton(
                    icon: const Icon(Icons.delete, color: Colors.red),
                    tooltip: "Eliminar Producto",
                    onPressed: () {
                      showDialog<void>(
                        context: context,
                        builder: (BuildContext dialogContext) {
                          return AlertDialog(
                            title: const Text("Confirmar Eliminación"),
                            content: Text(
                                "¿Está seguro que desea eliminar ${producto.nombre} (Código: ${producto.codigo})?"),
                            actions: <Widget>[
                              TextButton(
                                onPressed: () => Navigator.pop(dialogContext),

                                },

                          );

                            color: document.estadoDocumento == "Procesado (Completo)"
                                ? Colors.green.shade50

                                    ? Colors.red.shade50
                                    : Colors.grey.shade50,
                            borderRadius: BorderRadius.circular(4),

                                color: document.estadoDocumento == "Procesado (Completo)"

                                    : document.estadoDocumento == "Procesado (Incompleto)"
                                        ? Colors.red.shade200
                                        : Colors.grey.shade200),
                          ),
                          child: Text(
                            document.contenidoAlmacen.isNotEmpty ? document.contenidoAlmacen : "No procesado aún.",
                            style: const TextStyle(fontFamily: 'monospace'),
                          ),
                        ),
                      ],
                    ),
                  ),

                if (document.observacionAlmacen.isNotEmpty)
                  Padding(
                    padding: const EdgeInsets.symmetric(horizontal: 16.0, vertical: 4.0),
                    child: Align(
                      alignment: Alignment.centerLeft,
                      child: Text(
                        "Obs. Almacén: ${document.observacionAlmacen}",
                        style: const TextStyle(color: Colors.red, fontStyle: FontStyle.italic),

                if (document.fechaEnvioAlmacen != null)
                  Padding(
                    padding: const EdgeInsets.symmetric(horizontal: 16.0, vertical: 4.0),
                    child: Align(
                      alignment: Alignment.centerLeft,
                      child: Text(

                    ),

                if (document.fechaProcesamientoAlmacen != null)
                  Padding(
                    padding: const EdgeInsets.symmetric(horizontal: 16.0, vertical: 4.0),
                    child: Align(
                      alignment: Alignment.centerLeft,
                      child: Text(

                      ),
                    ),
                  ),
                const SizedBox(height: 8),
              ],
            ),
          ),
        ).toList(),
      ],
    );
  }

class _UserManagementTab extends StatefulWidget {
  @override
  State<_UserManagementTab> createState() => _UserManagementTabState();
}

class _UserManagementTabState extends State<_UserManagementTab> {
  final TextEditingController _addAgentNameController = TextEditingController();
  final TextEditingController _addAgentPasswordController = TextEditingController();

  @override
  void dispose() {
    _addAgentNameController.dispose();
    _addAgentPasswordController.dispose();
    super.dispose();
  }

  void _showAddAgentDialog(BuildContext context) {
    _addAgentNameController.clear();
    _addAgentPasswordController.clear();

    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: const Text("Agregar Nuevo Agente"),
          content: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: <Widget>[
                TextField(
                  controller: _addAgentNameController,
                  decoration: const InputDecoration(labelText: "Nombre del Agente"),
                ),
                const SizedBox(height: 10),
                TextField(
                  controller: _addAgentPasswordController,
                  decoration: const InputDecoration(labelText: "Contraseña del Agente"),
                  obscureText: true,
                ),
              ],
            ),
          ),
          actions: <Widget>[
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text("Cancelar"),
            ),
            ElevatedButton(
              onPressed: () {
                final String name = _addAgentNameController.text.trim();
                final String password = _addAgentPasswordController.text.trim();

                if (name.isEmpty || password.isEmpty) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(content: Text("Nombre y contraseña no pueden estar vacíos.")),
                  );
                  return;
                }

                final SistemaCentral sistema = Provider.of<SistemaCentral>(context, listen: false);
                if (sistema.getUser(name) != null) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(content: Text("Ya existe un usuario con el nombre '$name'.")),
                  );
                  return;
                }

                sistema.addUsuario(Usuario(nombre: name, rol: RolUsuario.agente, password: password));
                Navigator.pop(dialogContext);
                ScaffoldMessenger.of(context).showSnackBar(
                  SnackBar(content: Text("Agente '$name' agregado con éxito.")),
                );
              },
              child: const Text("Agregar"),
            ),
          ],
        );
      },
    );
  }

  void _showEditPasswordDialog(BuildContext context, Usuario user) {
    final TextEditingController passwordController = TextEditingController(text: user.password);

    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: Text("Editar Contraseña de ${user.nombre}"),
          content: TextField(
            controller: passwordController,
            obscureText: true,
            decoration: const InputDecoration(labelText: "Nueva Contraseña"),
          ),
          actions: <Widget>[
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text("Cancelar"),
            ),
            ElevatedButton(
              onPressed: () {
                final String newPassword = passwordController.text.trim();
                if (newPassword.isEmpty) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(content: Text("La contraseña no puede estar vacía.")),
                  );
                  return;
                }
                Provider.of<SistemaCentral>(context, listen: false)
                    .updateUsuarioPassword(user.nombre, newPassword);
                Navigator.pop(dialogContext);
                ScaffoldMessenger.of(context).showSnackBar(
                  SnackBar(content: Text("Contraseña de '${user.nombre}' actualizada.")),
                );
              },
              child: const Text("Guardar"),
            ),
          ],
        );
      },
    ).then((_) {
      passwordController.dispose();
    });
  }

  @override
  Widget build(BuildContext context) {
    final SistemaCentral sistema = Provider.of<SistemaCentral>(context);
    final List<Usuario> allUsers = sistema.allUsers;

    return Scaffold(
      body: ListView.builder(
        padding: const EdgeInsets.all(16),
        itemCount: allUsers.length,
        itemBuilder: (BuildContext context, int index) {
          final Usuario user = allUsers[index];
          return Card(
            margin: const EdgeInsets.only(bottom: 8),
            elevation: 2,
            child: ListTile(
              leading: Icon(
                user.rol == RolUsuario.admin
                    ? Icons.admin_panel_settings
                    : user.rol == RolUsuario.almacen
                        ? Icons.warehouse
                        : Icons.person,
                color: user.rol == RolUsuario.admin
                    ? Colors.red
                    : user.rol == RolUsuario.almacen
                        ? Colors.brown
                        : Colors.blue,
              ),
              title: Text(user.nombre, style: const TextStyle(fontWeight: FontWeight.bold)),
              subtitle: Text("Rol: ${user.rol.name}"),
              trailing: Row(
                mainAxisSize: MainAxisSize.min,
                children: <Widget>[
                  IconButton(
                    icon: const Icon(Icons.vpn_key, color: Colors.orange),
                    tooltip: "Editar Contraseña",
                    onPressed: () => _showEditPasswordDialog(context, user),
                  ),
                  if (user.rol == RolUsuario.agente)
                    IconButton(
                      icon: const Icon(Icons.delete, color: Colors.red),
                      tooltip: "Eliminar Agente",
                      onPressed: () {
                        showDialog<void>(
                          context: context,
                          builder: (BuildContext dialogContext) {
                            return AlertDialog(
                              title: const Text("Confirmar Eliminación"),
                              content: Text("¿Está seguro que desea eliminar al agente ${user.nombre}?"),
                              actions: <Widget>[
                                TextButton(
                                  onPressed: () => Navigator.pop(dialogContext),
                                  child: const Text("Cancelar"),
                                ),
                                ElevatedButton(
                                  style: ElevatedButton.styleFrom(backgroundColor: Colors.red),
                                  onPressed: () {
                                    sistema.removeUsuario(user.nombre);
                                    Navigator.pop(dialogContext);
                                    ScaffoldMessenger.of(context).showSnackBar(
                                      SnackBar(content: Text("Agente '${user.nombre}' eliminado.")),
                                    );
                                  },
                                  child: const Text("Eliminar"),
                                ),
                              ],
                            );
                          },
                        );
                      },
                    ),
                ],
              ),
            ),
          );
        },
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: () => _showAddAgentDialog(context),
        icon: const Icon(Icons.person_add),
        label: const Text("Agregar Agente"),
      ),
    );
  }
}
