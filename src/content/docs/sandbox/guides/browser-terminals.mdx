Build interactive terminal experiences by connecting xterm.js to sandbox shells via WebSocket. This guide shows you how to create browser-based terminals with automatic reconnection, output buffering, and multiple session support.

## Overview

Browser terminal integration involves three components:

1. **Cloudflare Worker** - Handles WebSocket upgrades and proxies to sandbox
2. **Sandbox container** - Runs the shell and manages terminal state
3. **Browser client** - xterm.js terminal with SandboxAddon for connection management

## Set up the Worker

Create a Worker that handles WebSocket connections and proxies them to sandbox terminals:

<TypeScriptExample>
```
import { getSandbox } from '@cloudflare/sandbox';

export { Sandbox } from "@cloudflare/sandbox";

interface Env {
Sandbox: DurableObjectNamespace;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    
    // Handle WebSocket upgrade for terminal connections
    if (request.headers.get('Upgrade')?.toLowerCase() === 'websocket') {
      if (url.pathname === '/ws/terminal') {
        return handleTerminal(request, env);
      }
      return new Response('Not found', { status: 404 });
    }
    
    // Serve your frontend HTML/assets
    if (url.pathname === '/') {
      return new Response(getTerminalHTML(), {
        headers: { 'Content-Type': 'text/html' }
      });
    }
    
    return new Response('Not found', { status: 404 });
  }
};

async function handleTerminal(request: Request, env: Env): Promise<Response> {
const url = new URL(request.url);
const sandboxId = url.searchParams.get('id');
const sessionId = url.searchParams.get('session');

if (!sandboxId) {
return new Response('Missing sandbox ID', { status: 400 });
}

const sandbox = getSandbox(env.Sandbox, sandboxId);

if (sessionId) {
// Connect to specific session
const session = await sandbox.getSession(sessionId);
return await session.terminal(request, { cols: 120, rows: 30 });
} else {
// Connect to default session  
 return await sandbox.terminal(request, { cols: 120, rows: 30 });
}
}

````
</TypeScriptExample>

### JavaScript implementation

<TypeScriptExample>
```javascript
import { Terminal } from '@xterm/xterm';
import { FitAddon } from '@xterm/addon-fit';
import { SandboxAddon } from '@cloudflare/sandbox/xterm';

class SandboxTerminalApp {
constructor() {
this.terminal = null;
this.addon = null;
this.currentSandboxId = 'user-terminal';
this.init();
}

init() {
this.setupTerminal();
this.setupUI();
this.connect();
}

setupTerminal() {
// Create xterm.js terminal
this.terminal = new Terminal({
cursorBlink: true,
fontSize: 14,
fontFamily: '"Cascadia Code", "Fira Code", Monaco, "Ubuntu Mono", monospace',
theme: {
background: '#000000',
foreground: '#ffffff',
cursor: '#ffffff',
selection: '#3e4451',
black: '#000000',
red: '#e06c75',
green: '#98c379',
yellow: '#e5c07b',
blue: '#61afef',
magenta: '#c678dd',
cyan: '#56b6c2',
white: '#ffffff',
brightBlack: '#5c6370',
brightRed: '#e06c75',
brightGreen: '#98c379',
brightYellow: '#e5c07b',
brightBlue: '#61afef',
brightMagenta: '#c678dd',
brightCyan: '#56b6c2',
brightWhite: '#ffffff'
}
});

    // Add fit addon for automatic sizing
    this.fitAddon = new FitAddon();
    this.terminal.loadAddon(this.fitAddon);

    // Create sandbox addon
    this.addon = new SandboxAddon({
      sandboxId: this.currentSandboxId,
      getWebSocketUrl: ({ sandboxId, sessionId }) => {
        const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
        const params = new URLSearchParams({ id: sandboxId });
        if (sessionId) params.set('session', sessionId);
        return `${protocol}//${window.location.host}/ws/terminal?${params}`;
      },
      reconnect: true,
      onStateChange: this.handleStateChange.bind(this)
    });

    this.terminal.loadAddon(this.addon);
    this.terminal.open(document.getElementById('terminal'));

    // Handle window resize
    window.addEventListener('resize', () => this.fitAddon.fit());
    this.fitAddon.fit();

}

setupUI() {
const sessionSelect = document.getElementById('session-select');
const reconnectBtn = document.getElementById('reconnect-btn');

    // Session switching
    sessionSelect.addEventListener('change', (e) => {
      const sessionId = e.target.value || undefined;
      this.addon.setSession(sessionId);
    });

    // Manual reconnect
    reconnectBtn.addEventListener('click', () => {
      this.addon.connect();
      reconnectBtn.style.display = 'none';
    });

}

connect() {
this.addon.connect();
}

handleStateChange(state, error) {
const statusEl = document.getElementById('connection-status');
const reconnectBtn = document.getElementById('reconnect-btn');

    // Update status indicator
    statusEl.className = `connection-status status ${state}`;

    switch (state) {
      case 'connecting':
        statusEl.textContent = 'Connecting...';
        reconnectBtn.style.display = 'none';
        break;

      case 'connected':
        statusEl.textContent = 'Connected';
        reconnectBtn.style.display = 'none';
        this.terminal.focus();
        break;

      case 'disconnected':
        if (error) {
          statusEl.textContent = `Error: ${error.message}`;
          reconnectBtn.style.display = 'inline-block';
        } else {
          statusEl.textContent = 'Disconnected';
        }
        break;
    }

    // Log for debugging
    console.log(`Terminal ${state}`, error);

}
}

// Initialize when DOM is ready
document.addEventListener('DOMContentLoaded', () => {
new SandboxTerminalApp();
});

````
</TypeScriptExample>

### Multiple terminals with tabs

Create a tabbed interface for multiple sessions:

<TypeScriptExample>
```javascript
class MultiTerminalApp {
  constructor() {
    this.terminals = new Map();
    this.activeTerminal = null;
    this.setupTabs();
  }

  createTerminal(sessionId = null) {
    const terminal = new Terminal(/* terminal options */);
    const fitAddon = new FitAddon();
    terminal.loadAddon(fitAddon);

    const addon = new SandboxAddon({
      sandboxId: 'multi-session-sandbox',
      sessionId,
      getWebSocketUrl: ({ sandboxId, sessionId }) => {
        const params = new URLSearchParams({ id: sandboxId });
        if (sessionId) params.set('session', sessionId);
        return `wss://my-worker.example.com/ws/terminal?${params}`;
      },
      onStateChange: (state, error) => {
        this.updateTabStatus(sessionId, state, error);
      }
    });

    terminal.loadAddon(addon);

    return { terminal, fitAddon, addon };
  }

  addTab(sessionId, label) {
    const terminalData = this.createTerminal(sessionId);
    this.terminals.set(sessionId, terminalData);

    // Create tab UI
    const tab = this.createTabElement(sessionId, label);
    document.getElementById('tabs').appendChild(tab);

    // Create terminal container
    const container = document.createElement('div');
    container.id = `terminal-${sessionId}`;
    container.className = 'terminal-pane';
    container.style.display = 'none';
    document.getElementById('terminal-container').appendChild(container);

    terminalData.terminal.open(container);
    terminalData.addon.connect();

    this.switchToTab(sessionId);
  }

  switchToTab(sessionId) {
    // Hide all terminals
    this.terminals.forEach((data, id) => {
      document.getElementById(`terminal-${id}`).style.display = 'none';
    });

    // Show selected terminal
    document.getElementById(`terminal-${sessionId}`).style.display = 'block';

    const terminalData = this.terminals.get(sessionId);
    terminalData.fitAddon.fit();
    terminalData.terminal.focus();

    this.activeTerminal = sessionId;
  }

  updateTabStatus(sessionId, state, error) {
    const tab = document.querySelector(`[data-session="${sessionId}"]`);
    if (tab) {
      tab.className = `tab ${state}`;
    }
  }
}
````

### Error handling and recovery

<TypeScriptExample>
```javascript
class RobustTerminal {
  constructor(options) {
    this.options = options;
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = 5;
    this.setup();
  }

  setup() {
    this.addon = new SandboxAddon({
      ...this.options,
      onStateChange: (state, error) => {
        this.handleConnectionState(state, error);
      }
    });
  }

  handleConnectionState(state, error) {
    switch (state) {
      case 'connected':
        this.reconnectAttempts = 0;
        this.showStatus('Connected', 'success');
        break;

      case 'connecting':
        this.showStatus('Connecting...', 'info');
        break;

      case 'disconnected':
        if (error) {
          this.handleError(error);
        } else {
          this.showStatus('Disconnected', 'warning');
        }
        break;
    }
  }

  handleError(error) {
    console.error('Terminal error:', error);

    if (error.message.includes('Session not found')) {
      this.showStatus('Session expired. Creating new session...', 'warning');
      // Let the addon handle reconnection
      return;
    }

    if (this.reconnectAttempts >= this.maxReconnectAttempts) {
      this.showStatus('Connection failed. Please refresh the page.', 'error');
      this.showReconnectButton();
      return;
    }

    this.reconnectAttempts++;
    this.showStatus(`Connection error. Retrying... (${this.reconnectAttempts}/${this.maxReconnectAttempts})`, 'error');
  }

  showStatus(message, type) {
    const statusEl = document.getElementById('status');
    statusEl.textContent = message;
    statusEl.className = `status ${type}`;
  }

  showReconnectButton() {
    document.getElementById('reconnect-btn').style.display = 'block';
  }
}
````

</TypeScriptExample>

### Monitor connection health

<TypeScriptExample>
```javascript
class TerminalMonitor {
  constructor(addon) {
    this.addon = addon;
    this.connectionTimes = [];
    this.errorCount = 0;
    this.setupMonitoring();
  }

  setupMonitoring() {
    let connectTime;

    this.addon.onStateChange = (state, error) => {
      switch (state) {
        case 'connecting':
          connectTime = Date.now();
          break;

        case 'connected':
          if (connectTime) {
            const duration = Date.now() - connectTime;
            this.connectionTimes.push(duration);
            this.reportMetric('connection_time', duration);
          }
          this.errorCount = 0;
          break;

        case 'disconnected':
          if (error) {
            this.errorCount++;
            this.reportMetric('connection_error', 1);
          }
          break;
      }
    };
  }

  reportMetric(name, value) {
    // Send to your analytics service
    console.log(`Metric: ${name} = ${value}`);
  }

  getStats() {
    const avgConnectionTime = this.connectionTimes.length > 0
      ? this.connectionTimes.reduce((a, b) => a + b) / this.connectionTimes.length
      : 0;

    return {
      avgConnectionTime,
      errorCount: this.errorCount,
      totalConnections: this.connectionTimes.length
    };
  }
}
````

async function ensureSession(sandbox, sessionId) {
  try {

  } catch (error) {
