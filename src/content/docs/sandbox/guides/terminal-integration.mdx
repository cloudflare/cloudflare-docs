Build interactive terminal interfaces for your sandboxes using the XTerm addon and PTY sessions.

## Overview

Terminal integration allows users to interact with sandboxes through familiar command-line interfaces. This is useful for:

- **Developer tools** - Code editors with integrated terminals
- **Educational platforms** - Interactive programming tutorials
- **Collaborative environments** - Shared terminal sessions
- **Remote access** - Web-based system administration

## Architecture

```
┌─────────────┐    WebSocket    ┌─────────────┐    PTY API    ┌─────────────┐
│  Browser    │◄─────────────►  │  Cloudflare │◄────────────► │  Sandbox    │
│ (xterm.js)  │                 │   Worker    │               │ Container   │
└─────────────┘                 └─────────────┘               └─────────────┘
       │                               │                             │
       ▼                               ▼                             ▼
   User Input                    WebSocket Handler                PTY Process
                                  Routes to sandbox              (/bin/bash)
```

## Basic terminal setup

### 1. Frontend (React)

Create a terminal component using xterm.js and the SandboxAddon:

<TypeScriptExample>
```
import { Terminal as XTerm } from '@xterm/xterm';
import { FitAddon } from '@xterm/addon-fit';
import { WebLinksAddon } from '@xterm/addon-web-links';
import { SandboxAddon } from '@cloudflare/sandbox/xterm';
import { useEffect, useRef, useState } from 'react';
import '@xterm/xterm/css/xterm.css';

interface TerminalProps {
sandboxId: string;
sessionId?: string;
}

export function Terminal({ sandboxId, sessionId }: TerminalProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [state, setState] = useState<'disconnected' | 'connecting' | 'connected'>('disconnected');

useEffect(() => {
if (!containerRef.current) return;

    const terminal = new XTerm({
      cursorBlink: true,
      fontFamily: '"JetBrains Mono", "Fira Code", monospace',
      fontSize: 14,
      theme: {
        background: '#1e1e1e',
        foreground: '#d4d4d4',
        cursor: '#ffffff'
      }
    });

    const fitAddon = new FitAddon();
    const webLinksAddon = new WebLinksAddon();
    const sandboxAddon = new SandboxAddon({
      getWebSocketUrl: ({ origin, sandboxId: sid, sessionId: session }) =>
        `${origin}/ws/terminal/${sid}/${session || 'default'}`,
      onStateChange: (newState) => setState(newState)
    });

    terminal.loadAddon(fitAddon);
    terminal.loadAddon(webLinksAddon);
    terminal.loadAddon(sandboxAddon);
    terminal.open(containerRef.current);
    fitAddon.fit();

    sandboxAddon.connect({ sandboxId, sessionId });

    const handleResize = () => fitAddon.fit();
    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('resize', handleResize);
      terminal.dispose();
    };

}, [sandboxId, sessionId]);

return (

<div className="relative h-full">
	<div ref={containerRef} className="h-full w-full" />
	{state === "connecting" && (
		<div className="absolute inset-0 flex items-center justify-center bg-black/50">
			<div className="text-white">Connecting...</div>
		</div>
	)}
</div>
); }

```
</TypeScriptExample>

### 2. Backend (Worker)

Handle WebSocket connections and create PTY sessions:

<TypeScriptExample>
```

import { getSandbox } from "@cloudflare/sandbox";

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    
    // Handle WebSocket upgrade for terminal connections
    if (url.pathname.startsWith('/ws/terminal/')) {
      return handleTerminalWebSocket(request, env);
    }
    
    return new Response('Not found', { status: 404 });
  }
};

async function handleTerminalWebSocket(request: Request, env: Env): Promise<Response> {
const upgrade = request.headers.get('Upgrade');
if (upgrade !== 'websocket') {
return new Response('Expected WebSocket', { status: 426 });
}

const url = new URL(request.url);
const [, , , sandboxId, sessionId] = url.pathname.split('/');

if (!sandboxId) {
return new Response('Missing sandbox ID', { status: 400 });
}

try {
const sandbox = getSandbox(env.Sandbox, sandboxId);
const session = await sandbox.getSession(sessionId || 'default');

    // The session.terminal() method handles the WebSocket upgrade
    return await session.terminal(request);

} catch (err) {
console.error('Terminal connection error:', err);
return new Response(
`Terminal error: ${err instanceof Error ? err.message : 'Unknown error'}`,
{ status: 500 }
);
}
}

```
</TypeScriptExample>

## Advanced patterns

### Multi-session support

Allow users to open multiple terminal sessions within the same sandbox:

<TypeScriptExample>
```

interface MultiTerminalProps {
sandboxId: string;
}

export function MultiTerminal({ sandboxId }: MultiTerminalProps) {
  const [sessions, setSessions] = useState<string[]>(['default']);
  const [activeSession, setActiveSession] = useState('default');

const addSession = () => {
const sessionId = `session-${Date.now()}`;
setSessions(prev => [...prev, sessionId]);
setActiveSession(sessionId);
};

const closeSession = (sessionId: string) => {
setSessions(prev => prev.filter(id => id !== sessionId));
if (activeSession === sessionId) {
setActiveSession(sessions[0] || 'default');
}
};

return (

<div className="h-full flex flex-col">
<div className="flex border-b">
{sessions.map(sessionId => (
<button
key={sessionId}
onClick={() => setActiveSession(sessionId)}
className={`px-3 py-2 ${activeSession === sessionId ? 'bg-blue-500' : 'bg-gray-200'}`} >
{sessionId}
{sessions.length > 1 && (
<span onClick={() => closeSession(sessionId)}>×</span>
)}
</button>
))}
<button onClick={addSession} className="px-3 py-2 bg-green-500"> + New Tab
</button>
</div>

      <div className="flex-1">
        <Terminal sandboxId={sandboxId} sessionId={activeSession} />
      </div>
    </div>

);
}

```
</TypeScriptExample>

### Session persistence

Preserve terminal state across browser refreshes:

<TypeScriptExample>
```

// Store session IDs in localStorage
const usePersistedSessions = (sandboxId: string) => {
const [sessions, setSessions] = useState<string[]>(() => {
const stored = localStorage.getItem(`sandbox-sessions-${sandboxId}`);
return stored ? JSON.parse(stored) : ['default'];
});

useEffect(() => {
localStorage.setItem(`sandbox-sessions-${sandboxId}`, JSON.stringify(sessions));

## Error handling

### Connection failures

Handle network issues and reconnection:

<TypeScriptExample>
```

const sandboxAddon = new SandboxAddon({
getWebSocketUrl: ({ origin, sandboxId, sessionId }) =>
`${origin}/ws/terminal/${sandboxId}/${sessionId || 'default'}`,

onStateChange: (state, error) => {
switch (state) {
case 'connecting':
setStatus('Connecting to terminal...');
break;

      case 'connected':
        setStatus('Terminal ready');
        setError(null);
        break;

      case 'disconnected':
        if (error) {
          setError(`Connection failed: ${error.message}`);
          if (error.message.includes('Max reconnection attempts')) {
            setStatus('Connection lost. Please refresh to retry.');
          } else {
            setStatus('Reconnecting...');
          }
        } else {
          setStatus('Disconnected');
        }
        break;
    }

},

reconnect: true // Enable automatic reconnection
});

```
</TypeScriptExample>

## Performance optimization

### Buffer management

For high-throughput applications, implement output buffering:

<TypeScriptExample>
```

class BufferedTerminalHandler {
private buffer: Uint8Array[] = [];
private flushTimer: ReturnType<typeof setTimeout> | null = null;

constructor(private webSocket: WebSocket) {}

write(data: Uint8Array) {
this.buffer.push(data);

    // Flush every 16ms (60fps) or when buffer gets large
    if (!this.flushTimer) {
      this.flushTimer = setTimeout(() => this.flush(), 16);
    } else if (this.buffer.length > 10) {
      this.flush();
    }

}

private flush() {
if (this.buffer.length === 0) return;

    // Combine all buffered data
    const totalLength = this.buffer.reduce((sum, arr) => sum + arr.length, 0);
    const combined = new Uint8Array(totalLength);
    let offset = 0;

    for (const chunk of this.buffer) {
      combined.set(chunk, offset);
      offset += chunk.length;
    }

    this.webSocket.send(combined);
    this.buffer = [];
    this.flushTimer = null;

}
}

```
</TypeScriptExample>
