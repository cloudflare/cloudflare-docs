In this guide, you will POST JavaScript your Worker did not ship and read the sandboxed execution result.

## Prerequisites

<Render file="prereqs" product="workers" />

## Create a Worker

<Steps>

1. Create a new Worker project:

   <PackageManagers
   	type="create"
   	pkg="cloudflare@latest"
   	args={"sandbox-dynamic-worker --category=hello-world --type=hello-world --lang=ts --no-deploy"}
   />

2. Change into the project directory:

   ```sh
   cd sandbox-dynamic-worker
   ```

</Steps>

## Add a Worker Loader binding

Add a [Worker Loader](/dynamic-workers/getting-started/#configure-worker-loader) binding so your Worker can call `load()`.

<WranglerConfig>

```jsonc
{
	"name": "sandbox-dynamic-worker",
	"main": "src/index.ts",
	"compatibility_date": "$today",
	"worker_loaders": [
		{
			"binding": "LOADER",
		},
	],
}
```

</WranglerConfig>

## Run posted JavaScript in the sandbox

Replace `src/index.ts`. The Worker reads `code` from the JSON body. That string becomes the body of `evaluate()`. A `return` statement is required.

`load()` does not pass `env`. `globalOutbound: null` blocks `fetch()` and `connect()`.

<TypeScriptExample filename="src/index.ts">

```ts
export default {
	async fetch(request: Request, env: Env): Promise<Response> {
		if (request.method !== "POST") {
			return new Response("POST JSON with a code string", { status: 405 });
		}

		const { code } = (await request.json()) as { code?: string };

		if (typeof code !== "string" || code.length === 0) {
			return new Response("JSON body must include a code string", {
				status: 400,
			});
		}

		const sandbox = env.LOADER.load({
			compatibilityDate: "$today",
			mainModule: "code.js",
			modules: {
				"code.js": `
					import { WorkerEntrypoint } from "cloudflare:workers";

					export class Code extends WorkerEntrypoint {
						evaluate() {
							${code}
						}
					}
				`,
			},
			globalOutbound: null,
		});

		const result = await sandbox.getEntrypoint("Code").evaluate();
		return Response.json({ result });
	},
};
```

</TypeScriptExample>

## Confirm the sandbox ran the posted code

<Steps>

1. Start a local session:

   <PackageManagers type="exec" pkg="wrangler" args="dev" />

2. POST JavaScript to the URL Wrangler printed. The default is `http://localhost:8787`:

   ```sh
   curl http://localhost:8787 --request POST --json '{"code":"const users = [{ name: \"Ada\", role: \"admin\" }, { name: \"Grace\", role: \"user\" }]; return users.filter((u) => u.role === \"admin\").map((u) => u.name);"}'
   ```

</Steps>

The response body should be `{"result":["Ada"]}`.
