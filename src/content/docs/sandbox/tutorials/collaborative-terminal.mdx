Build "Google Docs for bash" - a real-time collaborative terminal where multiple users can share the same PTY session and see each other's commands and output instantly.

**Time to complete:** 30 minutes

## What you'll build

A collaborative terminal application featuring:

- **Shared PTY sessions** - Multiple users connected to the same terminal process
- **Real-time synchronization** - All users see commands and output instantly  
- **User presence** - Visual indicators showing who's connected and typing
- **Room management** - Create and join terminal sessions via shareable links
- **Session persistence** - Terminal history preserved for new users joining

## 2. Understand the architecture

```
┌──────────────┐     WebSocket     ┌─────────────────┐     PTY API    ┌───────────┐
│   Browser    │◄──────────────────►  Cloudflare     │◄──────────────►│  Sandbox  │
│   (xterm.js) │                   │     Worker      │                │ Container │
└──────────────┘                   └─────────────────┘                └───────────┘
       │                                   │                                │
       │                                   │                                │
       ▼                                   ▼                                ▼
  User Input                         Broadcast PTY                     /bin/bash
  ─────────►                         output to all                          │
                                     connected users                        │
                                                                  ◄─────────▼
                                                                Real OS Process
```

The application uses three key components:

1. **Frontend React app** - Terminal UI with xterm.js and user management
2. **Cloudflare Worker** - Routes terminal WebSocket connections to sandbox sessions
3. **Room Durable Objects** - User presence and room state management

**Key architecture detail**: The terminal I/O flows directly between the browser and sandbox sessions via WebSocket, while room state (user presence, typing indicators) is managed separately by Durable Objects. This ensures optimal performance for terminal data while maintaining collaborative features.

## 3. Examine the frontend

The React application provides the terminal interface:

```ts title="app/components/Terminal.client.tsx"
import { SandboxAddon } from '@cloudflare/sandbox/xterm';
import { Terminal as XTerm } from '@xterm/xterm';
import { FitAddon } from '@xterm/addon-fit';

export function Terminal({ sandboxId, sessionId, onTyping }: TerminalProps) {
  useEffect(() => {
    const terminal = new XTerm({
      cursorBlink: true,
      fontFamily: '"JetBrains Mono", monospace',
      fontSize: 14
    });

    const fitAddon = new FitAddon();
    const sandboxAddon = new SandboxAddon({
      getWebSocketUrl: ({ origin, sessionId: sid }) =>
        `${origin}/ws/terminal/${sid}`,
      onStateChange: (state) => setState(state)
    });

    terminal.loadAddon(fitAddon);
    terminal.loadAddon(sandboxAddon);
    terminal.open(containerRef.current);
    fitAddon.fit();

    // Notify when user is typing
    terminal.onData(() => onTyping?.());
    
    // Connect to the shared session
    sandboxAddon.connect({ sandboxId, sessionId });

    return () => terminal.dispose();
  }, []);

  return <div ref={containerRef} className="h-full w-full" />;
}
```

Key features:
- **SandboxAddon** connects xterm.js to your Worker's WebSocket endpoint
- **onTyping callback** notifies other users when someone is typing
- **Automatic reconnection** handles network interruptions

## 4. Implement the backend

The Worker has two main responsibilities:

1. **Terminal routing** - Direct WebSocket connections between browsers and sandbox sessions
2. **Room management** - User presence and collaborative features via Durable Objects

```ts title="workers/app.ts"
// Direct terminal connection - high performance PTY I/O
if (url.pathname.startsWith('/ws/terminal/')) {
  const sessionId = url.pathname.split('/')[3];
  const sandbox = getSandbox(env.Sandbox, 'shared-terminal');
  const session = await sandbox.getSession(sessionId);
  return await session.terminal(request);
}

// Room management - user presence and state
if (url.pathname.startsWith('/ws/room/')) {
  const roomId = url.pathname.split('/')[3];
  const room = env.Room.get(env.Room.idFromName(roomId));
  return room.fetch(request);
}
```

The Room Durable Object handles collaborative features:

```ts title="workers/room.ts"
import { getSandbox } from '@cloudflare/sandbox';

export class Room extends DurableObject {
  private users = new Map<WebSocket, User>();

  async webSocketMessage(ws: WebSocket, message: string) {
    // Handle room-level control messages
    const msg = JSON.parse(message);
    
    if (msg.type === 'typing') {
      // Broadcast typing indicator to other users
      const user = this.users.get(ws);
      if (user) {
        this.broadcast({ type: 'user_typing', userId: user.id }, user.id);
      }
    }
  }

  async webSocketClose(ws: WebSocket) {
    const user = this.users.get(ws);
    this.users.delete(ws);
    
    this.broadcast({
      type: 'user_left',
      userId: user?.id,
      users: Array.from(this.users.values())
    });
  }

  private broadcast(message: any, excludeUserId?: string) {
    const data = JSON.stringify(message);
    this.users.forEach((user, ws) => {
      if (user.id !== excludeUserId && ws.readyState === WebSocket.READY_STATE_OPEN) {
        ws.send(data);
      }
    });
  }
}
    } else {
      // Handle control messages (resize, etc.)
      const msg = JSON.parse(message);
      if (msg.type === 'resize' && this.ptySession) {
        await this.ptySession.resize(msg.cols, msg.rows);
      }
    }
  }

  async webSocketClose(ws: WebSocket) {
    const user = this.users.get(ws);
    this.users.delete(ws);
    
    this.broadcast({
      type: 'user_left',
      userId: user?.id,
      users: Array.from(this.users.values())
    });

    // Clean up PTY if no users remain
    if (this.users.size === 0 && this.ptySession) {
      this.ptySession.destroy();
      this.ptySession = null;
    }
  }
```

Presence tracking with typing indicators:

```ts title="app/hooks/usePresence.ts"
export function usePresence({ roomId, userName }: UsePresenceOptions) {
  const [users, setUsers] = useState<User[]>([]);
  const [typingUsers, setTypingUsers] = useState<Set<string>>(new Set());
  const wsRef = useRef<WebSocket | null>(null);
  const typingTimeouts = useRef<Map<string, NodeJS.Timeout>>(new Map());

  useEffect(() => {
    const params = new URLSearchParams({ name: userName });
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const ws = new WebSocket(
      `${protocol}//${window.location.host}/ws/room/${roomId}?${params}`
    );
    
    ws.onmessage = (event) => {
      const message = JSON.parse(event.data);
      
      switch (message.type) {
        case 'user_joined':
          setUsers(message.users);
          break;
          
        case 'user_typing': {
          const userId = message.userId;
          setTypingUsers(prev => new Set(prev).add(userId));
          
          // Clear typing indicator after 2 seconds
          const existingTimeout = typingTimeouts.current.get(userId);
          if (existingTimeout) clearTimeout(existingTimeout);
          
          const timeout = setTimeout(() => {
            setTypingUsers(prev => {
              const next = new Set(prev);
              next.delete(userId);
              return next;
            });
            typingTimeouts.current.delete(userId);
          }, 2000);
          
          typingTimeouts.current.set(userId, timeout);
          break;
        }
      }
    };

    wsRef.current = ws;
    return () => ws.close();
  }, [roomId, userName]);

  const sendTyping = useCallback(() => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify({ type: 'typing' }));
    }
  }, []);

  return { users, typingUsers, sendTyping };
}
```

```ts
const rateLimiter = new Map<string, number>();

export async function createRoom(clientIP: string): Promise<string> {
  const now = Date.now();
  const lastCreated = rateLimiter.get(clientIP) || 0;
```

### Add file sharing

```ts
// Add file upload to your terminal room
async function handleFileUpload(file: File, roomId: string) {
  const sandbox = getSandbox(env.SANDBOX, 'shared-terminal');
  
  await sandbox.files.write(`/tmp/${file.name}`, await file.arrayBuffer());
  
  // Notify all users

    type: 'file_shared',
    filename: file.name,
```

### Session recording

Record terminal sessions for playback:

```ts
class SessionRecorder {
  private events: TerminalEvent[] = [];
  
  record(data: Uint8Array) {
    this.events.push({
      timestamp: Date.now(),
      type: 'output',
      data: Array.from(data)
    });
  }
  
  async saveRecording(roomId: string) {
    const recording = {
      roomId,
      startTime: this.events[0]?.timestamp || Date.now(),
      events: this.events
    };
    
    // Save to R2 or KV
    await env.RECORDINGS.put(`session-${roomId}`, JSON.stringify(recording));
  }
}
```

### Command history

Track and display command history:

```ts
class CommandTracker {
  private history: string[] = [];
  
  processInput(input: string) {
    // Detect command completion (Enter key)
    if (input.includes('\r') || input.includes('\n')) {
      const command = this.extractCommand(input);
      if (command) {
        this.history.push(command);
        this.broadcast({ type: 'command_history', commands: this.history });
      }
    }
  }
}
```
