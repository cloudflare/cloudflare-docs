Build a real-time collaborative terminal where multiple users can share the same shell session. Like Google Docs, but for your terminal.

This tutorial demonstrates PTY (pseudoterminal) integration, WebSocket communication, presence management, and room-based collaboration using Sandbox, Durable Objects, and xterm.js.

## Learning objectives

By the end of this tutorial, you will have:

- Built a multi-user terminal sharing application
- Integrated xterm.js with Sandbox PTY sessions
- Implemented real-time presence and typing indicators
- Created room-based collaboration with Durable Objects
- Set up WebSocket proxying for terminal I/O

## Before you start

- Complete the [Get Started guide](/sandbox/get-started/)
- Familiarity with [Durable Objects](/durable-objects/)
- [React](https://react.dev/) and [WebSocket](https://developer.mozilla.org/en-US/docs/Web/API/WebSocket) knowledge
- [Docker](https://www.docker.com/) installed and running

## 3. Define WebSocket protocol

Create the message types for room communication:

<TypeScriptExample>
```
// workers/types/protocol.ts

export interface User {
  id: string;
  name: string;
  color: string;
}

export interface RoomInfo {
  roomId: string;
  sessionId: string;
}

export type ConnectionState = 'disconnected' | 'connecting' | 'connected';

export type ServerMessage =
  | {
      type: 'connected';
      userId: string;
      user: User;
      users: User[];
      room: RoomInfo;
    }
  | { type: 'user_joined'; user: User; users: User[] }
  | { type: 'user_left'; userId: string; users: User[] }
  | { type: 'user_typing'; userId: string }
  | { type: 'room_deleted' }
  | { type: 'error'; message: string };

export type ClientMessage =
  | { type: 'typing' };

export interface ActiveRoom {
  roomId: string;
  userCount: number;
  createdAt: number;
}

export type RegistryMessage = { type: 'rooms'; rooms: ActiveRoom[] };
```
</TypeScriptExample>

## 4. Create Room Durable Object

The Room DO manages user presence and collaboration state:

<TypeScriptExample>
```
// workers/room.ts

import type { 
  User, 
  RoomInfo, 
  ServerMessage, 
  ClientMessage 
} from './types/protocol';

export class Room extends DurableObject {
  private users = new Map<WebSocket, User>();
  private roomInfo: RoomInfo;
  
  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env);
    
    // Each room maps to a unique sandbox session
    const roomId = this.ctx.id.toString();
    this.roomInfo = {
      roomId,
      sessionId: `room-${roomId}`
    };
  }

  async fetch(request: Request): Promise<Response> {
    if (request.headers.get('Upgrade') !== 'websocket') {
      return new Response('Expected WebSocket', { status: 426 });
    }

    const url = new URL(request.url);
    const userName = url.searchParams.get('name') || 'Anonymous';
    
    const { 0: client, 1: server } = new WebSocketPair();
    
    this.handleWebSocket(server, userName);
    
    return new Response(null, {
      status: 101,
      webSocket: client
    });
  }

  private handleWebSocket(ws: WebSocket, userName: string): void {
    const user: User = {
      id: crypto.randomUUID(),
      name: userName,
      color: this.generateUserColor()
    };
    
    this.users.set(ws, user);
    
    // Send initial state
    ws.send(JSON.stringify({
      type: 'connected',
      userId: user.id,
      user,
      users: Array.from(this.users.values()),
      room: this.roomInfo
    } satisfies ServerMessage));
    
    // Notify others of new user
    this.broadcast({
      type: 'user_joined',
      user,
      users: Array.from(this.users.values())
    }, ws);
    
    ws.addEventListener('message', (event) => {
      try {
        const message: ClientMessage = JSON.parse(event.data as string);
        this.handleMessage(ws, message);
      } catch {
        // Ignore malformed messages
      }
    });
    
    ws.addEventListener('close', () => {
      this.users.delete(ws);
      
      this.broadcast({
        type: 'user_left',
        userId: user.id,
        users: Array.from(this.users.values())
      });
      
      // Notify registry if room is empty
      if (this.users.size === 0) {
        this.notifyRegistryRoomEmpty();
      }
    });
    
    // Accept the WebSocket connection
    ws.accept();
    
    // Notify registry of active room
    this.notifyRegistryRoomActive();
  }

  private handleMessage(sender: WebSocket, message: ClientMessage): void {
    const user = this.users.get(sender);
    if (!user) return;
    
    switch (message.type) {
      case 'typing':
        this.broadcast({
          type: 'user_typing',
          userId: user.id
        }, sender);
        break;
    }
  }

  private broadcast(message: ServerMessage, exclude?: WebSocket): void {
    for (const [ws] of this.users) {
      if (ws !== exclude) {
        ws.send(JSON.stringify(message));
      }
    }
  }

  private generateUserColor(): string {
    const colors = [
      '#ef4444', '#f97316', '#eab308', '#22c55e',
      '#06b6d4', '#3b82f6', '#8b5cf6', '#ec4899'
    ];
    return colors[Math.floor(Math.random() * colors.length)];
  }

  private async notifyRegistryRoomActive(): Promise<void> {
    // Notify the registry that this room is active
    // Implementation depends on your registry setup
  }

  private async notifyRegistryRoomEmpty(): Promise<void> {
    // Notify the registry that this room is empty
    // Implementation depends on your registry setup
  }
}
```

## 5. Create Room Registry

The Registry DO tracks all active rooms globally:

<TypeScriptExample>
```
// workers/registry.ts

import type { ActiveRoom, RegistryMessage } from './types/protocol';

export class RoomRegistry extends DurableObject {
  private activeRooms = new Map<string, ActiveRoom>();
  private subscribers = new Set<WebSocket>();

  async fetch(request: Request): Promise<Response> {
    if (request.headers.get('Upgrade') === 'websocket') {
      return this.handleWebSocket(request);
    }

    const url = new URL(request.url);
    
    switch (request.method) {
      case 'POST':
        if (url.pathname === '/room-active') {
          const { roomId, userCount } = await request.json();
          return this.markRoomActive(roomId, userCount);
        }
        break;
      
      case 'DELETE':
        if (url.pathname.startsWith('/room/')) {
          const roomId = url.pathname.split('/').pop();
          return this.markRoomEmpty(roomId!);
        }
        break;
    }

    return new Response('Not found', { status: 404 });
  }

  private async handleWebSocket(request: Request): Promise<Response> {
    const { 0: client, 1: server } = new WebSocketPair();
    
    this.subscribers.add(server);
    
    // Send current rooms list
    server.send(JSON.stringify({
      type: 'rooms',
      rooms: Array.from(this.activeRooms.values())
    } satisfies RegistryMessage));
    
    server.addEventListener('close', () => {
      this.subscribers.delete(server);
    });
    
    server.accept();
    
    return new Response(null, {
      status: 101, 
      webSocket: client
    });
  }

  private async markRoomActive(roomId: string, userCount: number): Promise<Response> {
    this.activeRooms.set(roomId, {
      roomId,
      userCount,
      createdAt: Date.now()
    });
    
    this.broadcastRoomsUpdate();
    
    return new Response('OK');
  }

  private async markRoomEmpty(roomId: string): Promise<Response> {
    this.activeRooms.delete(roomId);
    this.broadcastRoomsUpdate();
    
    return new Response('OK');
  }

  private broadcastRoomsUpdate(): void {
    const message: RegistryMessage = {
      type: 'rooms',
      rooms: Array.from(this.activeRooms.values())
    };
    
    for (const ws of this.subscribers) {
      ws.send(JSON.stringify(message));
    }
  }
}
```
</TypeScriptExample>

## 6. Create main Worker

The main Worker routes requests and manages sandbox connections:

<TypeScriptExample>
```
// workers/app.ts

import { getSandbox } from '@cloudflare/sandbox';

export { Room } from './room';
export { RoomRegistry } from './registry';

interface Env {
  Room: DurableObjectNamespace;
  RoomRegistry: DurableObjectNamespace;
  Sandbox: DurableObjectNamespace;
}

export default {
  async fetch(request: Request, env: Env, ctx: ExecutionContext): Promise<Response> {
    const url = new URL(request.url);
    
    // WebSocket routes
    if (request.headers.get('Upgrade') === 'websocket') {
      if (url.pathname.startsWith('/ws/room/')) {
        const roomId = url.pathname.split('/').pop()!;
        const roomDO = env.Room.get(env.Room.idFromString(roomId));
        return roomDO.fetch(request);
      }
      
      if (url.pathname === '/ws/rooms') {
        const registryDO = env.RoomRegistry.get(env.RoomRegistry.idFromString('global'));
        return registryDO.fetch(request);
      }
      
      if (url.pathname.startsWith('/ws/terminal/')) {
        const sessionId = url.pathname.split('/').pop()!;
        const sandbox = getSandbox(env.Sandbox, 'shared-terminal');
        const session = await sandbox.getSession(sessionId);
        return session.terminal(request);
      }
    }
    
    // API routes
    if (url.pathname === '/api/room' && request.method === 'POST') {
      const roomId = crypto.randomUUID();
      
      // Create the session in advance
      const sandbox = getSandbox(env.Sandbox, 'shared-terminal');
      await sandbox.createSession({ id: `room-${roomId}` });
      
      return Response.json({ roomId });
    }
    
    if (url.pathname.startsWith('/api/room/') && request.method === 'DELETE') {
      const roomId = url.pathname.split('/').pop()!;
      
      // Notify room to close
      const roomDO = env.Room.get(env.Room.idFromString(roomId));
      await roomDO.fetch(new Request('http://localhost/delete', {
        method: 'POST'
      }));
      
      return new Response('OK');
    }
    
    // Serve static assets (React Router handles this in practice)
    return new Response('Not found', { status: 404 });
  }
};
```
</TypeScriptExample>

## 7. Build the terminal component

Create the React component that integrates xterm.js with the Sandbox:

<TypeScriptExample>
```
// app/components/Terminal.client.tsx

import { type ConnectionState, SandboxAddon } from '@cloudflare/sandbox/xterm';
import { FitAddon } from '@xterm/addon-fit';
import { WebLinksAddon } from '@xterm/addon-web-links';
import { Terminal as XTerm } from '@xterm/xterm';
import { useEffect, useRef, useState } from 'react';

import '@xterm/xterm/css/xterm.css';

interface TerminalProps {
  sandboxId: string;
  sessionId: string;
  onTyping?: () => void;
  onAddonReady?: (addon: SandboxAddon) => void;
}

export function Terminal({
  sandboxId,
  sessionId,
  onTyping,
  onAddonReady
}: TerminalProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [state, setState] = useState<ConnectionState>('disconnected');

  useEffect(() => {
    if (!containerRef.current) return;

    const terminal = new XTerm({
      cursorBlink: true,
      fontFamily: '"JetBrains Mono", "Fira Code", monospace',
      fontSize: 14,
      lineHeight: 1.2,
      theme: {
        background: '#09090b',
        foreground: '#fafafa',
        cursor: '#f97316',
        cursorAccent: '#09090b',
        selectionBackground: '#f9731640'
      }
    });

    const fitAddon = new FitAddon();
    const webLinksAddon = new WebLinksAddon();
    const sandboxAddon = new SandboxAddon({
      getWebSocketUrl: ({ origin, sessionId: sid }) =>
        `${origin}/ws/terminal/${sid}`,
      onStateChange: (newState) => setState(newState)
    });

    terminal.loadAddon(fitAddon);
    terminal.loadAddon(webLinksAddon);
    terminal.loadAddon(sandboxAddon);
    terminal.open(containerRef.current);
    fitAddon.fit();

    terminal.onData(() => onTyping?.());
    sandboxAddon.connect({ sandboxId, sessionId });
    onAddonReady?.(sandboxAddon);

    const handleResize = () => fitAddon.fit();
    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('resize', handleResize);
      terminal.dispose();
    };
  }, []);

  return (
    <div className="relative h-full">
      <div ref={containerRef} className="h-full w-full" />
      {state !== 'connected' && (
        <div className="absolute inset-0 flex items-center justify-center bg-zinc-950/80">
          <div className="flex items-center gap-3 text-zinc-400">
            {state === 'connecting' ? (
              <>
                <Spinner />
                <span>Connecting to terminal...</span>
              </>
            ) : (
              <span>Disconnected</span>
            )}
          </div>
        </div>
      )}
    </div>
  );
}

function Spinner() {
  return (
    <svg
      aria-hidden="true"
      className="w-5 h-5 animate-spin"
      viewBox="0 0 24 24"
      fill="none"
    >
      <circle
        className="opacity-25"
        cx="12"
        cy="12"
        r="10"
        stroke="currentColor"
        strokeWidth="4"
      />
      <path
        className="opacity-75"
        fill="currentColor"
        d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"
      />
    </svg>
  );
}
```
</TypeScriptExample>

interface UserAvatarsProps {

                style={{ backgroundColor: user.color }}
              />
            </span>

  );

## Architecture overview

Your collaborative terminal uses three WebSocket connections:

1. **Terminal I/O**: Browser ↔ Session PTY (via `session.terminal()`)
   - Raw bytes flow directly between xterm.js and the sandbox shell
   - No JSON protocol overhead for terminal data

2. **Room presence**: Browser ↔ Room Durable Object  
   - User join/leave events, typing indicators
   - Keeps collaboration state separate from terminal I/O

3. **Room registry**: Browser ↔ Registry Durable Object
   - Live updates of active rooms on homepage
   - Global room discovery and management

Each room maps to a unique session ID (`room-${roomId}`), providing isolation between different collaborative sessions within the same sandbox container.
